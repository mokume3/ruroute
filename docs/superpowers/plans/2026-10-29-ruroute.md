**Goal:** `docs/superpowers/specs/2026-10-29-ruroute-design.md` の仕様どおりに、型安全な URL を生成するルート定義ライブラリ ruroute を実装し、routopia との性能・バンドルサイズ比較を用意する。

```ts
import type { ParamValue, QueryValue } from './types'

export interface UrlArgs {
  params?: Record<string, ParamValue>
  query?: Record<string, QueryValue>
  hash?: ParamValue
}

/** baseUrl + パス + ?クエリ + #ハッシュ を組み立てる。値の検証はしない */
export function buildUrl(
  baseUrl: string | undefined,
  path: string,
  args: UrlArgs = {},
): string {
  const base = baseUrl?.replace(/\/+$/, '') ?? ''
  const pathname = path.replace(/\[([^\]]+)\]/g, (_, name: string) =>
    encodeURIComponent(String(args.params?.[name])),
  )

  const search = new URLSearchParams()
  for (const [key, value] of Object.entries(args.query ?? {})) {
    if (value === undefined) continue
    for (const v of Array.isArray(value) ? value : [value]) {
      search.append(key, String(v))
    }
  }
  const query = search.toString()

  const hash =
    args.hash === undefined ? '' : `#${encodeURIComponent(String(args.hash))}`

  return base + pathname + (query ? `?${query}` : '') + hash
}
```

```ts
/** type.* が返すマーカー。T は実行時には存在しないファントム型 */
export interface Schema<T = unknown> {
  readonly kind: string
  readonly [OUTPUT]: T
}

/** Schema から TS 型を取り出す */
export type Infer<S> = S extends Schema<infer T> ? T : never

/** type.empty */
export interface Empty {
  readonly kind: 'empty'
}

export type Method = 'get' | 'post' | 'put' | 'delete'

export type ParamsDef = Record<string, Schema<ParamValue>>
export type QueryDef = Record<string, Schema<QueryValue>>
export type HashDef = Schema<ParamValue | undefined>

export interface MethodDef {
  readonly params?: ParamsDef
  readonly query?: QueryDef
  readonly hash?: HashDef
}

export type MethodsDef = { readonly [M in Method]?: MethodDef | Empty }

/** type.route({...}) */
export interface Route<M extends MethodsDef = MethodsDef> {
  readonly kind: 'route'
  readonly methods: M
}

export type RouteDef = Route | Empty

/** type.route() の引数から、MethodsDef / MethodDef にないキー(未知メソッド・typo)を never にして弾く */
export type StrictMethods<M> = {
  [K in keyof M]: K extends Method
    ? M[K] extends Empty
      ? M[K]
      : M[K] & { [P in Exclude<keyof M[K], keyof MethodDef>]: never }
    : never
}
```

`src/type.ts`:

```ts
import type {
  Empty,
  Infer,
  MethodsDef,
  Route,
  Scalar,
  Schema,
  StrictMethods,
} from './types'

const schema = <T>(kind: string, extra?: object): Schema<T> =>
  ({ kind, ...extra }) as unknown as Schema<T>

/** パラメータを取らないルート・メソッド */
export const empty: Empty = { kind: 'empty' }

export const string: Schema<string> = schema('string')
export const number: Schema<number> = schema('number')
export const boolean: Schema<boolean> = schema('boolean')

/** 列挙した値のいずれか */
export const literal = <const T extends readonly (string | number)[]>(
  values: T,
): Schema<T[number]> => schema('literal', { values })

/** 要素の型のいずれか */
export const union = <const S extends readonly Schema<Scalar>[]>(
  schemas: S,
): Schema<Infer<S[number]>> => schema('union', { schemas })

/** 任意の TS 型(実行時は String() で文字列化) */
export const custom = <T extends Scalar>(): Schema<T> => schema('custom')

/** クエリ用の配列 */
export const array = <S extends Schema<Scalar>>(item: S): Schema<Infer<S>[]> =>
  schema('array', { item })

/** 省略可能 */
export const optional = <S extends Schema<Scalar | Scalar[]>>(
  inner: S,
): Schema<Infer<S> | undefined> => schema('optional', { inner })

/** ルート定義。引数なしは type.empty と同義 */
export function route(): Empty
export function route<const M extends MethodsDef>(
  methods: M & StrictMethods<M>,
): Route<M>
export function route(methods?: MethodsDef): Route | Empty {
  return methods === undefined ? empty : { kind: 'route', methods }
}
```

設計メモ: 置ける位置の制約は `ParamsDef` / `QueryDef` / `HashDef` が受け付ける `Schema<T>` の `T` で表現している(例: `Schema<boolean>` は `Schema<ParamValue>` に代入できない)。`union` / `array` / `optional` の引数制約も同じ仕組み。

```ts
// ---- 呼び出し引数 ----

type Simplify<T> = { [K in keyof T]: T[K] } & {}

type QueryArgs<Q extends QueryDef> = Simplify<
  {
    [K in keyof Q as undefined extends Infer<Q[K]> ? never : K]: Infer<Q[K]>
  } & {
    [K in keyof Q as undefined extends Infer<Q[K]> ? K : never]?: Infer<Q[K]>
  }
>

type HasRequiredKey<T> = Partial<T> extends T ? false : true

/** 正式名と短縮形のどちらか一方だけを受け付ける */
type Alias<
  Long extends string,
  Short extends string,
  V,
  Required extends boolean,
> = Required extends true
  ?
      | ({ [K in Long]: V } & { [K in Short]?: never })
      | ({ [K in Short]: V } & { [K in Long]?: never })
  :
      | ({ [K in Long]?: V } & { [K in Short]?: never })
      | ({ [K in Short]?: V } & { [K in Long]?: never })

type ParamsPart<D> = D extends { readonly params: infer P extends ParamsDef }
  ? Alias<'params', 'p', { [K in keyof P]: Infer<P[K]> }, true>
  : unknown

type QueryPart<D> = D extends { readonly query: infer Q extends QueryDef }
  ? Alias<'query', 'q', QueryArgs<Q>, HasRequiredKey<QueryArgs<Q>>>
  : unknown

type HashPart<D> = D extends { readonly hash: infer H extends HashDef }
  ? Alias<'hash', 'h', Infer<H>, undefined extends Infer<H> ? false : true>
  : unknown

/** メソッド定義 D の呼び出し引数(正式名・短縮形の両対応) */
export type MethodArgs<D> = ParamsPart<D> & QueryPart<D> & HashPart<D>

type HasAnyArg<D> = D extends
  | { readonly params: ParamsDef }
  | { readonly query: QueryDef }
  | { readonly hash: HashDef }
  ? true
  : false

/** メソッド定義 D の呼び出し関数。必須値がなければ引数を省略できる */
export type MethodFn<D> =
  HasAnyArg<D> extends false
    ? () => string
    : HasRequiredKey<MethodArgs<D>> extends true
      ? (args: MethodArgs<D>) => string
      : (args?: MethodArgs<D>) => string

// ---- ルートオブジェクト ----

type MethodsOf<R> = R extends Route<infer M> ? M : { readonly get: Empty }

export type RouteObject<R> = {
  readonly [K in keyof MethodsOf<R> & Method]: MethodFn<MethodsOf<R>[K]>
} & (MethodsOf<R> extends { readonly get: infer G } ? MethodFn<G> : unknown)

export type Routes<R> = { readonly [P in keyof R]: RouteObject<R[P]> }
```

```ts
import { buildUrl, type UrlArgs } from './build-url'
import type { RouteDef, Routes } from './types'

const METHODS = ['get', 'post', 'put', 'delete'] as const

interface CallArgs extends UrlArgs {
  p?: UrlArgs['params']
  q?: UrlArgs['query']
  h?: UrlArgs['hash']
}

const normalize = (args: CallArgs = {}): UrlArgs => ({
  params: args.params ?? args.p,
  query: args.query ?? args.q,
  hash: args.hash ?? args.h,
})

export interface RoutesConfig<R> {
  baseUrl?: string
  routes: R
}

export function defineRoutes<R extends Record<string, RouteDef>>(
  config: RoutesConfig<R>,
): Routes<R> {
  const result: Record<string, unknown> = {}
  const routes: Record<string, RouteDef> = config.routes
  for (const [path, def] of Object.entries(routes)) {
    const toUrl = (args?: CallArgs) =>
      buildUrl(config.baseUrl, path, normalize(args))
    const methods = def.kind === 'route' ? def.methods : { get: def }
    const fn = (args?: CallArgs) => toUrl(args)
    for (const method of METHODS) {
      if (method in methods) Object.assign(fn, { [method]: toUrl })
    }
    result[path] = fn
  }
  return result as Routes<R>
}
```

```ts
// ---- パス解析 ----

/** '/users/[id]/posts/[postId]' → 'id' | 'postId' */
export type PathParams<P extends string> =
  P extends `${string}[${infer Name}]${infer Rest}`
    ? Name | PathParams<Rest>
    : never

// ---- 定義の検証 ----

type ExpectedParams<P extends string> = {
  [K in PathParams<P>]: Schema<ParamValue>
}

type ValidateMethod<P extends string, D> = [PathParams<P>] extends [never]
  ? D extends Empty
    ? D
    : D & { readonly params?: undefined }
  : D extends Empty
    ? { readonly params: ExpectedParams<P> }
    : D extends { readonly params: infer Ps }
      ? D & {
          readonly params: {
            [K in keyof Ps]: K extends PathParams<P> ? unknown : never
          } & ExpectedParams<P>
        }
      : D & { readonly params: ExpectedParams<P> }

type ValidateRoute<P extends string, R> =
  R extends Route<infer M>
    ? Route<{ [K in keyof M]: ValidateMethod<P, M[K]> }>
    : ValidateMethod<P, R>

/** パスと params キーの一致を検証する。不一致のルートは型エラーになる */
export type ValidateRoutes<R> = {
  [P in keyof R]: P extends string ? ValidateRoute<P, R[P]> : never
}
```

`scripts/size.ts`:

```ts
import { readFileSync, rmSync } from 'node:fs'
import { join } from 'node:path'
import { brotliCompressSync, gzipSync } from 'node:zlib'
import { build } from 'tsdown'

const ENTRIES = {
  'ruroute (all exports)': 'bench/size/ruroute-all.ts',
  'routopia (all exports)': 'bench/size/routopia-all.ts',
  'ruroute (usage)': 'bench/size/ruroute-usage.ts',
  'routopia (usage)': 'bench/size/routopia-usage.ts',
}
const OUT_DIR = 'node_modules/.cache/size'

const rows: string[] = []
for (const [label, entry] of Object.entries(ENTRIES)) {
  const outDir = join(OUT_DIR, label.replace(/\W+/g, '-'))
  rmSync(outDir, { recursive: true, force: true })
  await build({
    config: false,
    entry: { index: entry },
    outDir,
    format: 'esm',
    platform: 'neutral',
    minify: true,
    dts: false,
    fixedExtension: true,
    deps: { alwaysBundle: [/./] },
    logLevel: 'silent',
  })
  const code = readFileSync(join(outDir, 'index.mjs'))
  rows.push(
    `| ${label} | ${code.length} | ${gzipSync(code, { level: 9 }).length} | ${brotliCompressSync(code).length} |`,
  )
}

console.log('| bundle | minified (B) | gzip (B) | brotli (B) |')
console.log('|---|---:|---:|---:|')
for (const row of rows) console.log(row)
```
